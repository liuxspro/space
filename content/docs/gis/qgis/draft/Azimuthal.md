        METHOD["Azimuthal Equidistant",
            ID["EPSG",1125]],

            ID["EPSG",8801]],
        PARAMETER["Longitude of natural origin",105,
            ANGLEUNIT["Degree",0.0174532925199433],
            ID["EPSG",8802]],
        PARAMETER["False easting",0,
            LENGTHUNIT["metre",1],
            ID["EPSG",8806]],

            LENGTHUNIT["metre",1],
            ID["EPSG",8807]]],
